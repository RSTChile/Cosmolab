  function gaussianBlur(src, dst, sigma) {
    if (sigma < 0.5) { dst.set(src); return; }
    const radius = Math.floor(sigma * 2), passes = 3;
    let buffer = new Float32Array(src), tmp = new Float32Array(SIZE * SIZE);
    for (let pass = 0; pass < passes; pass++) {
      for (let y = 0; y < SIZE; y++) for (let x = 0; x < SIZE; x++) {
        let sum = 0, count = 0;
        for (let dx = -radius; dx <= radius; dx++) {
          const nx = clamp(x + dx, 0, SIZE - 1);
          sum += buffer[idx(nx, y)]; count++;
        }
        tmp[idx(x, y)] = sum / count;
      }
      for (let y = 0; y < SIZE; y++) for (let x = 0; x < SIZE; x++) {
        let sum = 0, count = 0;
        for (let dy = -radius; dy <= radius; dy++) {
          const ny = clamp(y + dy, 0, SIZE - 1);
          sum += tmp[idx(x, ny)]; count++;
        }
        buffer[idx(x, y)] = sum / count;
      }
    }
    dst.set(buffer);
  }

        const v00 = src[idx(x0, y0)], v10 = src[idx(x1, y0)];
        const v01 = src[idx(x0, y1)], v11 = src[idx(x1, y1)];
        const v0 = v00 * (1 - fx) + v10 * fx, v1 = v01 * (1 - fx) + v11 * fx;
        dst[idx(x, y)] = v0 * (1 - fy) + v1 * fy;
      } else dst[idx(x, y)] = 0;
    }
  }

  function normalize(src, dst) {
    for (let i = 0; i < src.length; i++) dst[i] = clamp(src[i], 0, 1);
  }

  function iterate() {
    const { alpha, theta, sigma, beta, noise } = state.params;
    state.fieldPrev.set(state.field);
    rotate(state.fieldPrev, rotBuffer, theta);
    gaussianBlur(rotBuffer, blurBuffer, sigma);
    const temp = new Float32Array(SIZE * SIZE);
    for (let i = 0; i < SIZE * SIZE; i++) {
      const noiseVal = (Math.random() - 0.5) * 2 * noise;
      temp[i] = alpha * blurBuffer[i] + beta * state.fieldPrev[i] + noiseVal;
    }
    normalize(temp, state.field);
    if (document.getElementById('seedType').value === 'edge') {
      for (let i = 0; i < SIZE; i++) {
        state.field[idx(i, 0)] = 1.0; state.field[idx(i, SIZE-1)] = 1.0;
        state.field[idx(0, i)] = 1.0; state.field[idx(SIZE-1, i)] = 1.0;
      }
    }
    state.t++;
  }

  function computeMetrics() {
    let sum = 0, cx = 0, cy = 0;
    for (let i = 0; i < state.field.length; i++) {
      const v = state.field[i]; sum += v;
      const x = i % SIZE, y = Math.floor(i / SIZE);
      cx += x * v; cy += y * v;
    }
    const mean = sum / state.field.length;
    let edgeSum = 0;
    for (let y = 1; y < SIZE - 1; y++) for (let x = 1; x < SIZE - 1; x++) {
      const gx = -state.field[idx(x-1, y-1)] + state.field[idx(x+1, y-1)]
               - 2*state.field[idx(x-1, y)] + 2*state.field[idx(x+1, y)]
               - state.field[idx(x-1, y+1)] + state.field[idx(x+1, y+1)];
      const gy = -state.field[idx(x-1, y-1)] - 2*state.field[idx(x, y-1)] - state.field[idx(x+1, y-1)]
               + state.field[idx(x-1, y+1)] + 2*state.field[idx(x, y+1)] + state.field[idx(x+1, y+1)];
      edgeSum += Math.sqrt(gx*gx + gy*gy);
    }
    const ctxVal = (edgeSum / ((SIZE-2) * (SIZE-2) * 4)) * 100;
    if (sum > 0.01) { cx /= sum; cy /= sum; } else { cx = SIZE/2; cy = SIZE/2; }
    const n9eff = Math.sqrt(Math.pow(cx - SIZE/2, 2) + Math.pow(cy - SIZE/2, 2));
    let chiral = 0; const cx0 = SIZE / 2, cy0 = SIZE / 2;
    for (let y = 1; y < SIZE-1; y += 2) for (let x = 1; x < SIZE-1; x += 2) {
      const dx = x - cx0, dy = y - cy0;
      const gradX = (state.field[idx(x+1, y)] - state.field[idx(x-1, y)]) * 0.5;
      const gradY = (state.field[idx(x, y+1)] - state.field[idx(x, y-1)]) * 0.5;
      chiral += dx * gradY - dy * gradX;
    }
    chiral = chiral / (SIZE * SIZE / 4);
    return { ctx: clamp(ctxVal, 0, 100), out: clamp(mean * 100, 0, 100), chiral, n9eff };
  }

  function classifyRegime(m) {
    const { ctx, out, chiral } = m;
    if (out < 5) return 'Colapso';
    if (ctx < 10 && out > 70) return 'S.Cerrado';
    if (ctx > 80 && Math.abs(chiral) > 30) return 'Sobrecarga';
    if (ctx > 48 && ctx < 64 && out > 20 && out < 32 && Math.abs(chiral) > 0.5) return 'Zona Fértil';
    return 'Transición';
  }

  function render() {
    const cmap = COLORMAPS
    const cmap = COLORMAPS[document.getElementById('colormap').value];
    const data = imageData.data;
    for (let i = 0; i < state.field.length; i++) {
      const c = cmap(state.field[i]);
      const p = i * 4;
      data[p] = c[0]; data[p + 1] = c[1]; data[p + 2] = c[2]; data[p + 3] = 255;
    }
    ctx.putImageData(imageData, 0, 0);
  }

  function updateUI() {
    const m = computeMetrics();
    const regime = classifyRegime(m);
    document.getElementById('metricCTX').textContent = m.ctx.toFixed(1) + '%';
    document.getElementById('metricOUT').textContent = m.out.toFixed(1) + '%';
    document.getElementById('metricCHIRAL').textContent = m.chiral.toFixed(3);
    document.getElementById('metricN9').textContent = m.n9eff.toFixed(2);
    document.getElementById('timestep').textContent = state.t;
    const regimeEl = document.getElementById('regimeLabel');
    regimeEl.textContent = regime;
    regimeEl.className = 'text-lg font-bold ' + {
      'S.Cerrado': 'text-blue-400',
      'Zona Fértil': 'text-green-400',
      'Sobrecarga': 'text-yellow-400',
      'Colapso': 'text-red-400',
      'Transición': 'text-gray-400'
    }[regime];
    const isE11 = m.ctx >= 48 && m.ctx <= 64 && m.out >= 20 && m.out <= 32 && Math.abs(m.chiral) > 0.5;
    const validEl = document.getElementById('validationStatus');
    const cardEl = document.getElementById('validationCard');
    if (isE11) {
      validEl.textContent = '✓ Válido';
      validEl.className = 'text-xs px-2 py-1 rounded bg-green-600';
      cardEl.style.borderColor = '#10b981';
    } else {
      validEl.textContent = 'Inactivo';
      validEl.className = 'text-xs px-2 py-1 rounded bg-gray-700';
      cardEl.style.borderColor = '#1f2937';
    }
    document.getElementById('ctxTarget').textContent = m.ctx.toFixed(1) + '%';
    document.getElementById('outTarget').textContent = m.out.toFixed(1) + '%';
    document.getElementById('chiralTarget').textContent = m.chiral.toFixed(3);
    state.history.push({
      t: state.t,
     ...m,
     ...state.params,
      regime
    });
    if (state.history.length > state.maxHistory) state.history.shift();
    const labels = state.history.map(h => h.t);
    chart1.data.labels = labels;
    chart1.data.datasets[0].data = state.history.map(h => h.ctx);
    chart1.data.datasets[1].data = state.history.map(h => h.out);
    chart1.update('none');
    chart2.data.labels = labels;
    chart2.data.datasets[0].data = state.history.map(h => h.chiral);
    chart2.data.datasets[1].data = state.history.map(h => h.n9eff);
    chart2.update('none');
  }

  let animId = null;
  function loop() {
    if (!state.running) return;
    iterate();
    render();
    if (state.t % 5 === 0) {
      updateUI();
      if (state.t % 50 === 0) addTableRow();
    }
    animId = requestAnimationFrame(loop);
  }

  function start() {
    if (state.running) return;
    state.running = true;
    document.getElementById('btnStart').disabled = true;
    document.getElementById('btnPause').disabled = false;
    document.getElementById('runIndicator').className = 'w-2 h-2 rounded-full bg-green-500 running';
    loop();
  }

  function pause() {
    state.running = false;
    document.getElementById('btnStart').disabled = false;
    document.getElementById('btnPause').disabled = true;
    document.getElementById('runIndicator').className = 'w-2 h-2 rounded-full bg-gray-600';
    if (animId) cancelAnimationFrame(animId);
  }

  function reset() {
    pause();
    state.t = 0;
    state.history = [];
    initField(document.getElementById('seedType').value);
    render();
    updateUI();
    document.getElementById('resultsTable').innerHTML = '';
  }

    const h = state.history[state.history.length - 1];

      <td class="py-2 px-3">${h.t}</td>
      <td class="py-2 px-3">${h.alpha.toFixed(3)}</td>
      <td class="py-2 px-3">${h.theta.toFixed(2)}</td>
      <td class="py-2 px-3">${h.sigma.toFixed(1)}</td>

      <td class="py-2 px-3">${h.out.toFixed(1)}</td>

  }

  function exportCSV() {
    const headers = ['t','alpha','theta','sigma','beta','noise','CTX','OUT','CHIRAL','N9_eff','Regimen'];
    const rows = state.history.map(h => [
      h.t, h.alpha, h.theta, h.sigma, h.beta, h.noise,
      h.ctx.toFixed(3), h.out.toFixed(3), h.chiral.toFixed(6), h.n9eff.toFixed(3), h.regime
    ].join(','));
    const csv = [headers.join(','),...rows].join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url; a.download = `EIT3_results_t${state.t}.csv`; a.click();
    URL.revokeObjectURL(url);
  }

  function generatePhaseMap() {
    const canvas = document.getElementById('phaseCanvas');
    const ctx = canvas.getContext('2d');
    const w = canvas.width, h = canvas.height;
    const tempField = new Float32Array(SIZE * SIZE);
    const tempPrev = new Float32Array(SIZE * SIZE);
    const imgData = ctx.createImageData(w, h);
    const beta = state.params.beta;
    const noise = state.params.noise;
    const sigma = state.params.sigma;
    for (let iy = 0; iy < h; iy++) {
      for (let ix = 0; ix < w; ix++) {
        const alpha = ix / (w - 1);
        const theta = (1 - iy / (h - 1)) * 5;
        for (let y = 0; y < SIZE; y++) for (let x = 0; x < SIZE; x++)
          tempField[idx(x, y)] = gaussian(x, y, SIZE/2, SIZE/2, SIZE/8);
        tempField[idx(SIZE/2 + 6, SIZE/2)] += 0.02;
        for (let iter = 0; iter < 20; iter++) {
          tempPrev.set(tempField);
          rotate(tempPrev, rotBuffer, theta);
          gaussianBlur(rotBuffer, blurBuffer, sigma);
          for (let i = 0; i < SIZE * SIZE; i++) {
            tempField[i] = clamp(alpha * blurBuffer[i] + beta * tempPrev[i] + noise, 0, 1);
          }
        let sum = 0, edgeSum = 0;
        for (let i = 0; i < tempField.length; i++) sum += tempField[i];
        const out = (sum / tempField.length) * 100;
        for (let y = 1; y < SIZE - 1; y += 4) for (let x = 1; x < SIZE - 1; x += 4) {
          const gx = tempField[idx(x+1, y)] - tempField[idx(x-1, y)];
          const gy = tempField[idx(x, y+1)] - tempField[idx(x, y-1)];
          edgeSum += Math.sqrt(gx*gx + gy*gy);
        }
        const ctxVal = (edgeSum / ((SIZE/4) * (SIZE/4) * 4)) * 100;
        let color;
        if (out < 5) color = [239, 68, 68];
        else if (ctxVal < 10 && out > 70) color = [59, 130, 246];
        else if (ctxVal > 80) color = [245, 158, 11];
        else if (ctxVal > 48 && ctxVal < 64 && out > 20 && out < 32) color = [16, 185, 129];
        else color = [107, 114, 128];
        const p = (iy * w + ix) * 4;
        imgData.data[p] = color[0]; imgData.data[p + 1] = color[1]; imgData.data[p + 2] = color[2]; imgData.data[p + 3] = 255;
      }
    }
    ctx.putImageData(imgData, 0, 0);
    ctx.strokeStyle = '#fff'; ctx.lineWidth = 1; ctx.font = '10px JetBrains Mono'; ctx.fillStyle = '#fff';
    for (let i = 0; i <= 10; i++) {
      const x = i * w / 10, y = h - i * h / 10;
      ctx.fillText((i * 0.1).toFixed(1), x - 10, h + 12);
      ctx.fillText((i * 0.5).toFixed(1), -15, y + 3);
    }
    ctx.fillText('α', w / 2, h + 25);
    ctx.save(); ctx.translate(-25, h / 2); ctx.rotate(-Math.PI / 2); ctx.fillText('θ°', 0, 0); ctx.restore();
  }
